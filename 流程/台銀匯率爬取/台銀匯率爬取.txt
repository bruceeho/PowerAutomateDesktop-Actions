# ===== 台灣銀行今日牌告匯率 → 桌面 Excel =====
# 1. 準備檔名：桌面\台銀匯率_yyyyMMdd.xlsx
Folder.GetSpecialFolder SpecialFolder: Folder.SpecialFolder.DesktopDirectory SpecialFolderPath=> DesktopPath
DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> CurrentDateTime
Text.ConvertDateTimeToText.FromCustomDateTime DateTime: CurrentDateTime CustomFormat: $'''yyyyMMdd''' Result=> TodayText
# 2. 用 Chrome 開啟台銀牌告匯率頁面
WebAutomation.LaunchChrome.LaunchChrome Url: $'''https://rate.bot.com.tw/xrt?Lang=zh-TW''' WindowState: WebAutomation.BrowserWindowState.Maximized ClearCache: False ClearCookies: False WaitForPageToLoadTimeout: 60 Timeout: 60 PiPUserDataFolderMode: WebAutomation.PiPUserDataFolderModeEnum.AutomaticProfile TargetDesktop: $'''{"DisplayName":"Local Computer","Route":{"ServerType":"Local","ServerAddress":""}}''' BrowserInstance=> Browser
# 3. 讀取匯率表格（網站載入較慢時最多重試 15 次，每次間隔 2 秒）
SET RateCsv TO $''''''
LOOP LoopIndex FROM 1 TO 15 STEP 1
    WAIT 2
    WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() { var nl = String.fromCharCode(10); function clean(s) { var p = (s ? s : \"\").split(nl).map(function (x) { return x.trim(); }).filter(function (x) { return x.length; }); return p.length ? p[0] : \"\"; } var t = document.querySelector(\"table[title=牌告匯率]\"); if (!t) { t = document.querySelector(\"table.table\"); } if (!t) { return \"\"; } var rows = t.querySelectorAll(\"tbody tr\"); if (!rows.length) { return \"\"; } var tm = document.querySelector(\".time\"); var time = tm ? clean(tm.textContent) : \"\"; var keys = [\"本行現金買入\", \"本行現金賣出\", \"本行即期買入\", \"本行即期賣出\"]; var out = [\"幣別,現金買入,現金賣出,即期買入,即期賣出,掛牌時間\"]; Array.prototype.forEach.call(rows, function (r) { var c = r.querySelectorAll(\"td\"); if (!c[4]) { return; } var nm = r.querySelector(\".hidden-phone.print_show\"); if (!nm) { nm = c[0]; } var v = [clean(nm.textContent)]; keys.forEach(function (key, k) { var cell = r.querySelector(\"td[data-table=\" + key + \"]\"); if (!cell) { cell = c[k + 1]; } v.push(clean(cell.textContent)); }); v.push(time); out.push(v.join(\",\")); }); return out.join(nl); }''' Result=> RateCsv
    IF RateCsv <> $'''''' THEN
        EXIT LOOP
    END
END
WebAutomation.CloseWebBrowser BrowserInstance: Browser
IF RateCsv = $'''''' THEN
    Display.ShowMessageDialog.ShowMessage Title: $'''台銀匯率''' Message: $'''讀取不到匯率表格，請確認網路或台銀網站是否正常。''' Icon: Display.Icon.Warning Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> ButtonPressed
    EXIT Code: 0
END
# 4. 轉成資料表（第一列是標題）
Variables.GenerateDataTableFromCSV.ReadPredefinedSeparator CSVText: RateCsv TrimFields: True FirstLineContainsColumnNames: False ReadFieldsAsText: False ColumnsSeparator: Variables.CSVColumnsSeparator.Comma CSVTable=> RateTable
# 5. 寫入 Excel 並存到桌面
Excel.LaunchExcel.Launch Visible: False UseMachineLocale: False LoadAddInsAndMacros: False Instance=> ExcelInstance
Excel.WriteToExcel.WriteCell Instance: ExcelInstance Value: RateTable Column: $'''A''' Row: 1
Excel.RenameWorksheet.RenameWorksheetWithIndex Instance: ExcelInstance Index: 1 NewName: TodayText
Excel.ResizeColumnsOrRows.AutofitAllColumns Instance: ExcelInstance
Excel.CloseExcel.CloseAndSaveAs Instance: ExcelInstance DocumentFormat: Excel.ExcelFormat.FromExtension DocumentPath: $'''%DesktopPath%\\台銀匯率_%TodayText%.xlsx'''
Display.ShowMessageDialog.ShowMessage Title: $'''台銀匯率''' Message: $'''已儲存：%DesktopPath%\\台銀匯率_%TodayText%.xlsx''' Icon: Display.Icon.Information Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> ButtonPressed
