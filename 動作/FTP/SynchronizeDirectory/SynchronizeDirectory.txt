#   DeleteIfSourceAbsent [布林，預設 False]
#   IncludeSubdirectories [布林，預設 True]
#   TimeDifferenceHours [整數，預設 0]
#   TimeDifferenceMinutes [整數，預設 0]

#   FilesDeleted [List<FileObject>]
